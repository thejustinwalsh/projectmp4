cmake_minimum_required(VERSION 3.20)
project(deck_loops VERSION 0.1.0 LANGUAGES C CXX)
add_library(deck_loops SHARED src/deck_loops.cpp)
target_compile_features(deck_loops PRIVATE cxx_std_17)
target_compile_definitions(deck_loops PRIVATE DECK_LOOPS_BUILD)
target_include_directories(deck_loops PUBLIC include)
set_target_properties(deck_loops PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
  target_compile_options(deck_loops PRIVATE /W4 /WX /fp:precise)
else()
  target_compile_options(deck_loops PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(loop_contract tests/contract.c)
  target_link_libraries(loop_contract PRIVATE deck_loops)
  target_compile_features(loop_contract PRIVATE c_std_11)
  if(MSVC)
    target_compile_options(loop_contract PRIVATE /UNDEBUG)
  else()
    target_compile_options(loop_contract PRIVATE -UNDEBUG)
  endif()
  add_test(NAME loop_c_abi_contract COMMAND loop_contract)
endif()
